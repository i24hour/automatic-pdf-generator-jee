\begin{enumerate}[leftmargin=*,itemsep=2pt]
    \item This question paper contains \textbf{\VAR{total_questions}} questions.
    \item The question paper is divided into subject-wise sections.
    \BLOCK{ if level == 'Mains' or level == 'JEE Mains' }
    \item \textbf{SECTION A - MCQs:} Each question has 4 options with ONLY ONE correct answer.
    \begin{itemize}[itemsep=1pt]
        \item Correct Answer: \textbf{+4 marks}
        \item Wrong Answer: \textbf{-1 mark}
        \item Unattempted: \textbf{0 marks}
    \end{itemize}
    \item \textbf{SECTION B - Numerical Value:} Answer should be rounded off to 2 decimal places.
    \begin{itemize}[itemsep=1pt]
        \item Correct Answer: \textbf{+4 marks}
        \item Wrong Answer: \textbf{0 marks}
        \item Unattempted: \textbf{0 marks}
    \end{itemize}
    \BLOCK{ elif level == 'NEET' }
    \item Each question carries \textbf{4 marks} for correct answer.
    \item \textbf{1 mark will be deducted} for every wrong answer.
    \item No marks will be deducted for unattempted questions.
    \BLOCK{ elif level == 'Advanced' or level == 'JEE Advanced' }
    \item \textbf{Single Correct MCQs:} +3 for correct, -1 for wrong.
    \item \textbf{Multiple Correct MCQs:} +4 for all correct, +1 for partial, -2 for wrong.
    \item \textbf{Numerical Value:} +3 for correct, 0 for wrong.
    \BLOCK{ else }
    \item Each correct MCQ carries \textbf{+4 marks}, wrong answer \textbf{-1 mark}.
    \item Numerical questions: \textbf{+4 marks} for correct, \textbf{0 marks} for wrong.
    \BLOCK{ endif }
    \item Use of calculator is NOT permitted.
    \item Rough work may be done on blank pages provided.
